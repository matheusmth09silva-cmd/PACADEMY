/* ---------------- TIPOGRAFIA: divisão em linhas e palavras -------------- */
function dividirPalavras(el){
if (el.dataset.dividido) return el.querySelectorAll('.palavra');
var texto = el.textContent;
el.textContent = '';
texto.split(/(\s+)/).forEach(function(parte){
if (!parte) return;
var span = document.createElement('span');
span.className = 'palavra';
span.textContent = parte;
el.appendChild(span);
});
el.dataset.dividido = '1';
return el.querySelectorAll('.palavra');
}

// Agrupa as palavras por linha visual e envolve cada linha numa máscara
function dividirLinhas(el){
var palavras = dividirPalavras(el);
var linhas = [], atual = null, topoAtual = null;
palavras.forEach(function(w){
var topo = Math.round(w.offsetTop);
if (topoAtual === null || Math.abs(topo - topoAtual) > 4) {
topoAtual = topo;
atual = [];
linhas.push(atual);
}
atual.push(w);
});
var frag = document.createDocumentFragment();
linhas.forEach(function(grupo){
var mask = document.createElement('span');
mask.className = 'linha-mask';
var inner = document.createElement('span');
grupo.forEach(function(w){ inner.appendChild(w); });
mask.appendChild(inner);
frag.appendChild(mask);
});
el.innerHTML = '';
el.appendChild(frag);
return el.querySelectorAll('.linha-mask > span');
}
